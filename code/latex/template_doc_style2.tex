%    所以这里只框「输入」（题目/定理），不框「输出」（解答过程）。

%  正文由 render.py 注入到文件末尾的占位符处，请勿手动改动

% 【题目框】蓝框，标题左贴 —— 用来框「题目/问题」
\newtcolorbox{problembox}[1][题目]{
    enhanced, breakable,
    colback=ptbg, colframe=ptblue!70!black,
    title={\color{white}\bfseries #1},
    attach boxed title to top left={xshift=3mm,yshift=-3mm},
    boxed title style={colback=ptblue,sharp corners,size=small},
    fonttitle=\large,
    before skip=12pt, after skip=14pt
}

% 【定理框】红框，标题居中 —— 用来框「定理/引理/结论」
\newtcolorbox{theorembox}[1][定理]{
    enhanced, breakable,
    colback=ptbg, colframe=ptred!70!black,
    title={\color{white}\bfseries #1},
    attach boxed title to top center={xshift=0mm,yshift=-3mm},
    boxed title style={colback=ptred,sharp corners,size=small},
    fonttitle=\large\centering,
    before skip=16pt, after skip=14pt
}

% 【注意框】橙框 —— 可选，用来强调易错点
\newtcolorbox{notebox}[1][注意]{
    enhanced, breakable,
    colback=ptbg, colframe=ptorange!80!black,
    title={\color{white}\bfseries #1},
    attach boxed title to top left={xshift=3mm,yshift=-3mm},
    boxed title style={colback=ptorange,sharp corners,size=small},
    fonttitle=\large,
    before skip=12pt, after skip=12pt
}
